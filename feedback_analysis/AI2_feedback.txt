# Example of antithetic variates in Monte Carlo
def generate_paths_antithetic(num_sims, days_to_expiry, drift, volatility, current_price):
    # We'll generate half the number of simulations and then use antithetic variates for the other half
    half_sims = num_sims // 2
    dt = 1/252
    paths = np.zeros((half_sims * 2, days_to_expiry))
    paths[:, 0] = current_price

    for t in range(1, days_to_expiry):
        # Generate random shocks for half the simulations
        z = np.random.standard_normal(half_sims)
        # Use antithetic variates for the other half
        z_antithetic = -z

        # Combine the two sets of shocks
        all_z = np.concatenate([z, z_antithetic])

        # Apply the Brownian motion
        paths[:, t] = paths[:, t-1] * np.exp(
            (drift - 0.5 * volatility**2) * dt +
            volatility * np.sqrt(dt) * all_z
        )

    return paths

def enhance_liquidity_screening(self, option_chain: List) -> List:
    """Ensure tradeability with liquidity metrics"""
    screened_options = []

    for option in option_chain:
        # Bid-ask spread as % of mid price
        mid_price = (option['bid'] + option['ask']) / 2
        spread_pct = (option['ask'] - option['bid']) / mid_price if mid_price > 0 else 1.0

        # Volume and open interest thresholds
        volume_ok = option['volume'] > 1000
        oi_ok = option['open_interest'] > 5000
        spread_ok = spread_pct < 0.10  # Max 10% spread

        if volume_ok and oi_ok and spread_ok:
            option['liquidity_score'] = min(1.0, (option['volume'] / 10000) * 0.5 +
                                                   (option['open_interest'] / 20000) * 0.3 +
                                                   (1 - spread_pct) * 0.2)
            screened_options.append(option)

    return sorted(screened_options, key=lambda x: x['liquidity_score'], reverse=True)
